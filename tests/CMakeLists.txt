# Test executables registered with CTest. No TIMEOUT is set anywhere: tests run
# naturally; a hang is diagnosed as a lifecycle defect, not masked.
set(ER_TEST_TARGETS)

function(er_add_test name)
  add_executable(${name} ${ARGN})
  target_link_libraries(${name} PRIVATE er_core)
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  if(ER_BUILD_TRANSPORT)
    target_link_libraries(${name} PRIVATE er_transport)
  endif()
  add_test(NAME ${name} COMMAND ${name})
  list(APPEND ER_TEST_TARGETS ${name})
endfunction()

er_add_test(er_test_core test_core.cpp)
er_add_test(er_test_property test_property.cpp)
er_add_test(er_test_concurrency test_concurrency.cpp)
er_add_test(er_test_adversarial test_adversarial.cpp)

if(ER_BUILD_TRANSPORT)
  add_executable(er_test_integration test_integration.cpp spawn.cpp)
  target_link_libraries(er_test_integration PRIVATE er_core er_transport)
  target_include_directories(er_test_integration PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  add_test(NAME er_test_integration COMMAND er_test_integration)
endif()
if(ER_BUILD_CUDA)
  add_executable(er_test_cuda test_cuda.cpp)
  target_link_libraries(er_test_cuda PRIVATE er_core er_cuda)
  if(WIN32)
    target_link_libraries(er_test_cuda PRIVATE ws2_32)
  endif()
  add_test(NAME er_test_cuda COMMAND er_test_cuda)
endif()
